/* ========================================================= 接口配置：按你 _worker.js 里的真实路径改这里 ========================================================= */ const API = { list: "/api/stations", // 电台列表 search: (kw) => `/api/search?kw=${encodeURIComponent(kw)}`, }; /* ---------- 状态 ---------- */ const state = { all: [], region: "全部", category: "全部", keyword: "" }; /* ---------- 工具 ---------- */ function escapeHtml(s){ return String(s==null?"":s) .replace(/&/g,"&").replace(//g,">") .replace(/"/g,""").replace(/'/g,"'"); } async function request(url){ const resp = await fetch(url, { headers:{ "Accept":"application/json" } }); const text = await resp.text(); try { return JSON.parse(text); } catch { throw new Error("返回不是 JSON：" + text.slice(0,120)); } } /* ---------- 渲染 ---------- */ const gridEl = document.getElementById("grid"); const titleEl = document.getElementById("title"); const countEl = document.getElementById("count"); function setStatus(t){ gridEl.innerHTML = `
${escapeHtml(t)}
`; } function renderFilters(){ const regions = ["全部", ...new Set(state.all.map(s => s.region).filter(Boolean))]; const cats = ["全部", ...new Set(state.all.map(s => s.category).filter(Boolean))]; const row = (label, items, key, current) => `
${label}
${items.map(v => `${escapeHtml(v)}`).join("")}
`; document.getElementById("filters").innerHTML = row("地区", regions, "region", state.region) + row("分类", cats, "category", state.category); } function renderGrid(){ const list = state.all.filter(s => { const okR = state.region === "全部" || s.region === state.region; const okC = state.category === "全部" || s.category === state.category; const okK = !state.keyword || (s.name||"").includes(state.keyword) || (s.sub||"").includes(state.keyword); return okR && okC && okK; }); countEl.textContent = `共 ${list.length} 个电台`; if (!list.length){ setStatus("没有找到符合条件的电台"); return; } gridEl.innerHTML = list.map(s => { const cover = s.cover ? `` : `
${escapeHtml(s.name||"电台").slice(0,6)}
`; return `
${cover}
直播中
${escapeHtml(s.name)}
${escapeHtml(s.sub||"")}
`; }).join(""); } /* ---------- 交互 ---------- */ function setFilter(key, value){ state[key] = value; renderFilters(); renderGrid(); } function doSearch(){ state.keyword = document.getElementById("kw").value.trim(); renderGrid(); } function playStation(id){ const s = state.all.find(x => String(x.id) === String(id)); if (!s) return; const url = s.streamUrl || s.url || s.src; if (!url){ alert("该电台暂无播放地址"); return; } const player = document.getElementById("player"); const audio = document.getElementById("audio"); document.getElementById("pTitle").textContent = s.name || "—"; document.getElementById("pSub").textContent = s.sub || ""; audio.src = url; player.classList.add("show"); audio.play().catch(()=>{}); } function closePlayer(){ const audio = document.getElementById("audio"); audio.pause(); audio.src = ""; document.getElementById("player").classList.remove("show"); } /* ---------- 数据归一化：按 _worker.js 的真实返回结构改 ---------- */ function normalize(data){ const arr = Array.isArray(data) ? data : (data?.data || data?.list || data?.stations || []); return arr.map(x => ({ id: x.id || x.radioId || x.radio_id || x.name, name: x.name || x.title || x.radioName || "未命名", cover: x.cover || x.image || x.logo || x.pic || "", sub: x.sub || x.desc || x.description || x.current || "", region: x.region || x.province || "", category: x.category || x.type || "", streamUrl: x.streamUrl || x.url || x.src || x.playUrl || "", })); } /* ---------- 启动 ---------- */ async function boot(){ setStatus("加载中…"); try { const data = await request(API.list); state.all = normalize(data); renderFilters(); renderGrid(); } catch(e){ setStatus("加载失败：" + e.message); } } document.getElementById("kw").addEventListener("keydown", e => { if (e.key === "Enter") doSearch(); }); window.setFilter = setFilter; window.doSearch = doSearch; window.playStation = playStation; window.closePlayer = closePlayer; boot();
